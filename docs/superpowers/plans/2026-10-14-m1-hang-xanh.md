# M1 Hàng Xanh — Kế hoạch thực thi

> **Cho agent thực thi:** làm từng task theo thứ tự phụ thuộc; mỗi bước dùng checkbox `- [ ]`. Bạn chỉ thấy task của mình; phần "Interfaces" là hợp đồng bắt buộc — dùng đúng tên hàm/kiểu đã ghi. Không tự đổi quyết định; thấy mâu thuẫn thì ghi vào báo cáo cuối phiên.

**Mục tiêu:** M1 là lát mỏng xuyên suốt ở cụm Hàng Xanh: app=smart trên cảnh 3D (khóa khu `bac`), đèn thích ứng max-pressure có công tắc A/B + KPI đo được, phát hiện xe vượt đèn đỏ có bằng chứng ảnh, backend + hai web tối thiểu để người dân tra cứu, và script chạy/đo toàn hệ.

**Kiến trúc:** tab 3D (`app=smart`) là nguồn sự thật — sim thuần JS tất định (`city-traffic-sim-31.js` đã sửa để nhận `signalPlan/controller/breakerS`) sinh số đo, vi phạm, ảnh bằng chứng, khung stream rồi đẩy lên backend FastAPI + SQLite qua REST/WS (hàng đợi IndexedDB khi mất mạng). Hai web React chỉ hiển thị. Mọi module mới là `*-32.js`, nối vào app 31A bằng tham số tùy chọn; không truyền tham số mới thì hành vi 31A y nguyên (QA 31 vẫn chạy).

**Công nghệ:** Node 24 (`node --test` cho logic sim/đèn/luật), Python 3 + venv riêng (FastAPI, uvicorn, bcrypt, PyJWT, Pydantic, pytest), SQLite (WAL), React + Vite (một project hai entry: officer, citizen), Playwright/Chromium kiểu `outputs/hcmc-poc/scripts/qa-*-31.cjs` (biến `PW_MODULE`, `PW_CHROME`), Three.js r186 WebGPU + WebGL2 fallback, IndexedDB cho hàng đợi trạm.

```js
// group_axis_deg: hướng trục đường chính tại nút (độ, 0–180, trong hệ cảnh x–z).
// Quy tắc suy trục: đường cấp cao nhất qua nút (trunk > primary > …); cùng cấp thì
// tổng chiều dài cạnh qua nút lớn nhất; trục = hướng trung bình modulo 180° của các
// nhánh thuộc đường đó (đã tra trên road-graph-31.json).
export const SIGNAL_PLAN_32_VERSION = 'signal-plan-32a';

export const SIGNAL_PLAN_32 = [
  { node_id: 'HX-01', osm_id: 366390689, group_axis_deg: 160.4, offset_s: 0 },
  { node_id: 'HX-02', osm_id: 366440827, group_axis_deg: 17.6, offset_s: 8 },
  { node_id: 'HX-03', osm_id: 10118771782, group_axis_deg: 4.3, offset_s: 16 },
  { node_id: 'HX-04', osm_id: 12437015526, group_axis_deg: 93.4, offset_s: 24 },
  { node_id: 'HX-05', osm_id: 701671562, group_axis_deg: 162.9, offset_s: 0 },
  { node_id: 'HX-06', osm_id: 1497276515, group_axis_deg: 171.4, offset_s: 32 }
];

const CLUSTER_M = 80;

export function buildSignalPlan32({ graph, nodeMap }) {
  const index = new Map(graph.nodes.map((n, i) => [n.id, i]));
  const rowByOsm = new Map(SIGNAL_PLAN_32.map(r => [r.osm_id, r]));
  const metaByOsm = new Map(nodeMap.map(m => [m.osm_id, m]));
  const signalNodes = [], offset = new Map();
  // Cụm: hai nút cách < CLUSTER_M thì chung cụm; id theo nodeMap (HX-C1…).
  const clusters = [];
  const clusterByOsm = new Map();
  for (const row of SIGNAL_PLAN_32) {
    const meta = metaByOsm.get(row.osm_id);
    if (!meta) throw new Error(`node_map.json thiếu osm_id ${row.osm_id}`);
    let cluster = clusters.find(c => c.cluster_id === meta.cluster_id);
    if (!cluster) {
      cluster = { cluster_id: meta.cluster_id, nodeIds: [], offset_s: row.offset_s };
      clusters.push(cluster);
    }
    const ni = index.get(row.osm_id);
    if (ni === undefined) throw new Error(`graph thiếu nút ${row.osm_id}`);
    cluster.nodeIds.push(ni);
    clusterByOsm.set(row.osm_id, cluster);
    signalNodes.push(ni);
    offset.set(ni, cluster.offset_s);
  }
  // arriveGroup theo st = edge*2 + (dir>0 ? 0 : 1): nhóm 0 nếu |cos(hướng vào − trục)| > 0,5.
  const arriveGroup = new Uint8Array(graph.edges.length * 2);
  const E = graph.edges;
  for (let st = 0; st < arriveGroup.length; st++) {
    const e = st >> 1, dir = st & 1 ? -1 : 1;
    const toId = dir > 0 ? E[e].to : E[e].from;
    const row = rowByOsm.get(toId);
    if (!row) continue;
    const c = E[e].c, n = c.length / 2;
    const head = dir > 0
      ? Math.atan2(c[n * 2 - 1] - c[n * 2 - 3], c[n * 2 - 2] - c[n * 2 - 4])
      : Math.atan2(c[3] - c[1], c[2] - c[0]);
    arriveGroup[st] = Math.abs(Math.cos(head - row.group_axis_deg * Math.PI / 180)) > 0.5 ? 0 : 1;
  }
  return { signalNodes, offset, arriveGroup, clusters };
}
```

```js
export const CONTROLLER_32_VERSION = 'controller-32a';
export const SIGNAL_PARAMS_32 = Object.freeze({
  yellowS: 3, allRedS: 2, minGreenS: 8, maxGreenS: 45, decisionIntervalS: 1, switchMargin: 0.15,
  fixed: Object.freeze({ green0S: 22, green1S: 16 }), clusterM: 80
});

const GREEN = 'green', YELLOW = 'yellow', RED = 'red';

export function createSignalController32({ net, sim = null, mode = 'fixed', params = {}, redRun = null }) {
  if (mode !== 'fixed' && mode !== 'smart') throw new TypeError('mode phải là fixed|smart');
  let P = { ...SIGNAL_PARAMS_32, ...params, fixed: { ...SIGNAL_PARAMS_32.fixed, ...(params.fixed || {}) } };
  if (!(P.minGreenS < P.maxGreenS)) throw new TypeError('params: minGreenS phải < maxGreenS');
  let curMode = mode, pendingMode = null;
  let simRef = sim;
  const normNode = node => (typeof node === 'string' ? net.G.nodes.findIndex(x => String(x.id) === node) : node);
  // Cụm đèn: đơn vị quyết định. clusterOf: nodeIndex → id cụm.
  const clusters = [], clusterOf = new Map();
  if (net.signalPlan && net.signalPlan.clusters) {
    for (const c of net.signalPlan.clusters) {
      const st = { cluster_id: c.cluster_id, nodeIds: [...c.nodeIds], offset_s: c.offset_s, group: 0, phase: GREEN, tPhase: 0, cycle: 0, switches: 0, lastDecision: -Infinity };
      clusters.push(st);
      for (const n of c.nodeIds) clusterOf.set(n, st);
    }

    const st = { cluster_id: 'NET-C1', nodeIds: [], offset_s: 0, group: 0, phase: GREEN, tPhase: 0, cycle: 0, switches: 0, lastDecision: -Infinity };
    for (let n = 0; n < net.signal.length; n++) if (net.signal[n]) { st.nodeIds.push(n); clusterOf.set(n, st); }
    clusters.push(st);
  }
  const nodeIdOf = n => String(net.G.nodes[n]?.id ?? n);
  const overrides = new Map(); // nodeIndex → {group, by, reason, until_t_sim, until_at}
  // Lịch sử KPI: mẫu mỗi 1 s trong cửa sổ 60 s.

  const samples = []; // {t, wait_avg, nodeWait: Map, nodeQueue: Map, clusterQueue: Map}
  let switchWindow = [];
  const greenOf = c => c.group === 0 ? P.fixed.green0S : P.fixed.green1S;
  const phaseLen = (c, phase) => phase === GREEN ? (curMode === 'fixed' ? greenOf(c) : c.greenLen ?? P.minGreenS) : phase === YELLOW ? P.yellowS : P.allRedS;
  const advancePhase = (c, tSim) => {
    if (c.phase === RED) {
      if (pendingMode) { curMode = pendingMode; pendingMode = null; for (const x of clusters) x.greenLen = P.minGreenS; }
      c.cycle += 1;
      c.group = c.group === 0 ? 1 : 0;
      c.phase = GREEN;
      c.greenLen = P.minGreenS;
      c.lastDecision = tSim;
      c.switches++;
      switchWindow.push(tSim);
    } else if (c.phase === GREEN) c.phase = YELLOW;
    else c.phase = RED;
    c.tPhase = tSim;
  };
  const pressure = (c, queues, g) => {
    let p = 0;
    for (const n of c.nodeIds) {
      const rows = queues.get(n);
      if (!rows) continue;
      for (const r of rows) {
        if (r.group !== g) continue;
        p += r.count - Math.min(r.count, r.crossed || 0);
      }
    }
    return p;
  };
  const decision = (c, tSim, queues) => {
    if (curMode !== 'smart' || c.phase !== GREEN) return;
    const elapsed = tSim - c.tPhase;
    if (elapsed < P.minGreenS) return;
    if (elapsed - (c.lastDecision - c.tPhase) < P.decisionIntervalS && elapsed < P.maxGreenS) return;
    c.lastDecision = tSim;
    const p0 = pressure(c, queues, 0), p1 = pressure(c, queues, 1);
    const total = Math.max(1, p0 + p1), margin = Math.abs(p0 - p1) / total;
    const want = p0 > p1 ? 0 : 1;
    if (elapsed >= P.maxGreenS || (margin >= P.switchMargin && want !== c.group)) { c.greenLen = elapsed; advancePhase(c, tSim); }
    else if (c.greenLen === undefined) c.greenLen = Math.min(P.maxGreenS, Math.max(P.minGreenS, elapsed));

    const o = overrides.get(node);
    if (!o) return null;
    if (tSim > o.until_t_sim) { overrides.delete(node); return null; }
    return o;
```

```js
const COLORS = ['trắng', 'đen', 'xám', 'bạc', 'đỏ', 'xanh dương', 'xanh lá', 'vàng', 'nâu', 'cam', 'hồng', 'tím'];
const OWNERS = [];   // điền từ backend/app/seed/owners.json khi chạy trên Node; mặc định 200 tên trong seed.

export function mulberry32(a) {
  return function () {
    a |= 0; a = (a + 0x6D2B79F5) | 0;
    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}
export const hashWord = rnd => Math.floor(rnd() * 4294967296) >>> 0;

export const SCENARIO_PLATES_32 = [
  { slot: 1, gen: 0, plate: '51K-888.88', vehicle_class: 'car' },
  { slot: 2, gen: 0, plate: '59-X9 999.99', vehicle_class: 'motorbike' },
  { slot: 3, gen: 0, plate: '51K-111.11', vehicle_class: 'car' },
  { slot: 4, gen: 0, plate: '59-A1 222.22', vehicle_class: 'motorbike' },
  { slot: 5, gen: 0, plate: '51G-333.33', vehicle_class: 'car' },
  { slot: 6, gen: 0, plate: '59-B2 444.44', vehicle_class: 'motorbike' }
];

export function loadOwners32(names) { OWNERS.length = 0; OWNERS.push(...names); }

const fmtOf = cls => cls === 'motorbike' ? 'motorbike' : 'car';

export function plateFor32(slot, gen, vehicleClass) {
  const r0 = hashU32((SEED ^ slot ^ (gen << 16)) >>> 0);
  const rnd = mulberry32(r0);
  // 1) biển — đúng 4 lần rút cho (biển, màu, chủ, điện thoại)
  let b = hashWord(rnd);
  const letter = SERIES[b % 21]; b = Math.floor(b / 21);
  let plate;
  const plate_format = fmtOf(vehicleClass);
  if (plate_format === 'motorbike') {
    const d = b % 10; b = Math.floor(b / 10);
    const n = b % 1000; b = Math.floor(b / 1000);
    const p = b % 100;
    plate = `59${letter}${d} ${String(n).padStart(3, '0')}.${String(p).padStart(2, '0')}`;
  } else {
    const withDigit = b % 2; b = Math.floor(b / 2);
    let mid = '';
    if (withDigit) { mid = String(b % 10); b = Math.floor(b / 10); }
    const n = b % 1000; b = Math.floor(b / 1000);
    const p = b % 100;
    plate = `51${letter}${mid}-${String(n).padStart(3, '0')}.${String(p).padStart(2, '0')}`;
  }
  const k2 = hashWord(rnd);
  const color = COLORS[Math.floor(k2 / 4294967296 * 12)];   // = (k2 * 12) >> 32
  const k3 = hashWord(rnd);
  const owner_name = OWNERS.length ? OWNERS[Math.floor(k3 / 4294967296 * OWNERS.length)] : `Chủ xe ${slot}`;   // = (k3 * n) >> 32
  const k4 = hashWord(rnd);
  const owner_phone = '09' + String(k4 % 100000000).padStart(8, '0');
  const sc = SCENARIO_PLATES_32.find(s => s.slot === slot && s.gen === gen && fmtOf(s.vehicle_class) === plate_format);
  return sc
    ? { slot, gen, plate: sc.plate, plate_format, vehicle_class: vehicleClass, color, owner_name, owner_phone, is_scenario: true }
    : { slot, gen, plate, plate_format, vehicle_class: vehicleClass, color, owner_name, owner_phone, is_scenario: false };
}

const CONFUSE = { '0': '8', '8': '0', '1': '7', '7': '1', '5': 'S', 'S': '5', 'B': '8', 'D': '0', 'I': '1' };

export function ocrNoise32(plateTrue, distanceM, rangeM, tSim, slot, gen) {
  const rnd = mulberry32(hashU32((SEED ^ slot ^ (gen << 16) ^ Math.floor(tSim)) >>> 0));
  let swaps = 0;
  const chars = [];
  for (const ch of plateTrue) {
    const uMiss = rnd(), uSwap = rnd();
    if (uMiss < 0.02) continue;                    // p_miss = 0,02: mất 1 ký tự
    if (uSwap < 0.06 && CONFUSE[ch]) { chars.push(CONFUSE[ch]); swaps++; }
    else chars.push(ch);
  }
  const plate_read = chars.join('');
  const ocr_confidence = Math.max(0, Math.min(1, 0.98 - 0.12 * (distanceM / rangeM) - 0.25 * swaps));
  return { plate_read, ocr_confidence, swaps };
}

export function createPerception32({ net, sim, controller, nodeMap, cameras, seed = SEED }) {
  const { lanes, Rz, arriveGroup, G, E } = net;
  const idxByOsm = new Map(G.nodes.map((n, i) => [n.id, i]));
  const metaByNode = new Map(nodeMap.map(m => [idxByOsm.get(m.osm_id), m]));
  const speedLimitOf = l => {
    const road = G.roads[E[lanes.edge[l]].road];
    const max = road && road.maxspeed ? Number(road.maxspeed) : NaN;
    if (Number.isFinite(max)) return max;
    const t = String(road?.type || '').replace(/_link$/, '');
    return { trunk: 60, primary: 50, secondary: 40, tertiary: 30, residential: 30 }[t] ?? 30;
  };
  // Máy trạng thái theo (xe, camera): watch → grace → violation/idle.
  const states = new Map();     // key `${i}|${camera_id}` → {phase, tRed, inFovS, distanceM}
  const seen = new Set();       // `${vehicle_id}|${node_id}|${cycle_id}` đã sinh
  const out = [];
  const detections = new Map();
  const angleDiff = (a, b) => { let d = b - a; while (d > Math.PI) d -= 2 * Math.PI; while (d < -Math.PI) d += 2 * Math.PI; return d; };
  const yawRad = c => c.yaw_deg * Math.PI / 180;
  const rec0Of = i => plateFor32(i + 1, sim.gen[i], TRAFFIC_KINDS[sim.kind[i]].name);

  const scan = tSim => {
    for (const cam of cameras) {
      const nodeIdx = idxByOsm.get(nodeMap.find(m => m.node_id === cam.node_id)?.osm_id);
      const frameDetections = [];
      for (let i = 0; i < sim.N; i++) {
        if (!sim.active[i]) continue;
        const l = sim.lane[i];
        const node = lanes.to[l];
        const stopPos = lanes.length[l] - (Rz[node] || 0) - sim.len[i] / 2;
        // Vị trí xe theo pose-lite: nội suy dọc polyline cạnh (đủ cho kiểm FOV mặt phẳng XZ).
        const e = lanes.edge[l], c = E[e].c, n = c.length / 2;
        const x = c[0] + (c[n * 2 - 2] - c[0]) * (sim.s[i] / Math.max(0.01, lanes.length[l]));
        const z = c[1] + (c[n * 2 - 1] - c[1]) * (sim.s[i] / Math.max(0.01, lanes.length[l]));
        const dx = cam.x - x, dz = cam.z - z;
        const distance = Math.hypot(dx, dz);
        const toCam = Math.atan2(dz, dx);
        const off = Math.abs(angleDiff(yawRad(cam), toCam));
        const inFov = distance <= cam.range_m && off <= (cam.fov_deg * Math.PI / 180) / 2;
        const key = `${i}|${cam.camera_id}`;
        let st = states.get(key);
        if (!st) { st = { phase: 'idle', tRed: 0, inFovS: 0, distanceM: distance }; states.set(key, st); }
        if (inFov) st.inFovS += 0.5; else { st.phase = 'idle'; continue; }
        const group = arriveGroup[lanes.st[l]];
        const green = controller.isGreen(node, lanes.st[l], tSim);
        const crossed = sim.s[i] > stopPos;
        if (!green && !crossed) {
          if (st.phase === 'idle') { st.phase = 'grace'; st.tRed = tSim; }
        } else if (green) st.phase = 'idle';
        const rec0 = rec0Of(i);
        if (st.phase === 'grace' && crossed && tSim - st.tRed > GRACE_S) {
          st.phase = 'idle';
          const meta = metaByNode.get(node);
          const kindName = TRAFFIC_KINDS[sim.kind[i]].name;
          const slot = i + 1, gen = sim.gen[i];
          const ocr = ocrNoise32(rec0.plate, distance, cam.range_m, tSim, slot, gen);
          const cycle_id = controller.cycleIdOf(node, tSim);
          const dedupe = `${slot}|${meta?.node_id ?? node}|${cycle_id}`;
          if (seen.has(dedupe)) continue;
          seen.add(dedupe);
          const confidence = Math.max(0, Math.min(1,
            0.5 * ocr.ocr_confidence + 0.3 * (1 - distance / cam.range_m) + 0.2 * (st.inFovS >= 1 ? 1 : 0)));
          const phaseFor = controller.stateOf(node, tSim).phases.find(p => p.group === group);
          const carLike = kindName !== 'motorbike';
          out.push({
            violation_code: 'RED-LIGHT',
            violation_label: 'Không chấp hành hiệu lệnh của đèn tín hiệu giao thông',
            status: confidence >= 0.85 ? 'new' : 'needs_review',
            priority_flag: false, priority_reason: null,
            plate: rec0.plate, plate_format: rec0.plate_format, vehicle_class: kindName,
            vehicle_color: rec0.color, vehicle_id: `V-${String(slot).padStart(6, '0')}`,
            owner_name: rec0.owner_name, owner_phone: rec0.owner_phone, is_scenario_plate: rec0.is_scenario,
            occurred_at: null, node_id: meta?.node_id ?? String(G.nodes[node].id),
            node_osm_id: G.nodes[node].id, node_label: meta?.label ?? '',
            camera_id: cam.camera_id, road_index: lanes.edge[l], lane_index: l,
            signal: {
              mode: controller.getMode(), cluster_id: controller.stateOf(node, tSim).cluster_id,
              cycle_id, phase_for_vehicle: phaseFor ? phaseFor.phase : 'red', group,
              elapsed_in_phase_s: +(tSim - st.tRed).toFixed(1), cycle_s: controller.stateOf(node, tSim).cycle_s,
              speed_kmh: +(sim.v[i] * 3.6).toFixed(1), speed_limit_kmh: speedLimitOf(l)
            },
            confidence: +confidence.toFixed(3), ocr_confidence: +ocr.ocr_confidence.toFixed(3),
            detection: { in_fov_s: +st.inFovS.toFixed(1), distance_m: +distance.toFixed(1), grace_s: GRACE_S, crossed_stop_line_at_s: +tSim.toFixed(1) },
            evidence: [],
            penalty: carLike
              ? { vnd_min: 18000000, vnd_max: 20000000, license_deduction_points: 4, source: 'NĐ 168/2024 · Điều 6 · khoản 9 · điểm b', verified: true }
              : { vnd_min: 4000000, vnd_max: 6000000, license_deduction_points: 4, source: 'NĐ 168/2024 · Điều 7 · khoản 7 · điểm c', verified: true },
            sim: { t_sim: +tSim.toFixed(1), seed, scenario: 'peak-hang-xanh', station_id: 'STATION-3D-01', station_time: new Date().toISOString() },
            review: { by: null, at: null, note: null }, complaint_ids: []
          });
        }
        frameDetections.push({ vehicle_id: `V-${String(i + 1).padStart(6, '0')}`, kind: rec0.vehicle_class, color: rec0.color, distance_m: +distance.toFixed(1), in_fov_s: +st.inFovS.toFixed(1) });
      }
      detections.set(cam.camera_id, frameDetections);
    }
  };
  return {
    scan,
    drainViolations: () => { const list = out.slice(); out.length = 0; return list; },
    detectionsOf: cameraId => detections.get(cameraId) || [],
    stateHash: () => hashU32(out.length * 7919 + seen.size * 131 + (sim.counters.step | 0)).toString(16).padStart(8, '0')
  };
}
```

```python
"""Bộ sinh biển số/xe/chủ tất định — bản Python song song plateFor32 (violation-perception-32.js).

Chạy: python seed_plates.py --out plates.json
Thuật toán phải cho KẾT QUẢ GIỐNG HỆT bản JS (test chéo seed-plates-32.test.mjs).

import argparse
import json
import os

SEED_PLATES = 31031
SERIES = 'ABCDEFGHKLMNPRSTUVWXYZ'
COLORS = ['trắng', 'đen', 'xám', 'bạc', 'đỏ', 'xanh dương', 'xanh lá', 'vàng', 'nâu', 'cam', 'hồng', 'tím']
SCENARIO_PLATES = [
    {'slot': 1, 'gen': 0, 'plate': '51K-888.88', 'vehicle_class': 'car'},
    {'slot': 2, 'gen': 0, 'plate': '59-X9 999.99', 'vehicle_class': 'motorbike'},

    {'slot': 6, 'gen': 0, 'plate': '59-B2 444.44', 'vehicle_class': 'motorbike'},
]


def hash_u32(n):
    x = (n ^ 0x9E3779B9) & 0xFFFFFFFF
    x = ((x ^ (x >> 16)) * 0x85EBCA6B) & 0xFFFFFFFF
    x = ((x ^ (x >> 13)) * 0xC2B2AE35) & 0xFFFFFFFF
    return (x ^ (x >> 16)) & 0xFFFFFFFF


def mulberry32(a0):
    """Trả về hàm rút word uint32 — khớp mulberry32 bản JS (r() = word / 2**32)."""
    a = a0 & 0xFFFFFFFF

    def nxt():
        nonlocal a
        a = (a + 0x6D2B79F5) & 0xFFFFFFFF
        t1 = ((a ^ (a >> 15)) * (1 | a)) & 0xFFFFFFFF
        inner = ((t1 ^ (t1 >> 7)) * (61 | t1)) & 0xFFFFFFFF
        t2 = (((t1 + inner) & 0xFFFFFFFF) ^ t1) & 0xFFFFFFFF
        return (t2 ^ (t2 >> 14)) & 0xFFFFFFFF

    return nxt


def plate_for(slot, gen, vehicle_class, owners):
    r0 = hash_u32((SEED_PLATES ^ slot ^ ((gen << 16) & 0xFFFFFFFF)) & 0xFFFFFFFF)
    rnd = mulberry32(r0)
    b = rnd()
    letter = SERIES[b % 21]
    b //= 21
    fmt = 'motorbike' if vehicle_class == 'motorbike' else 'car'
    if fmt == 'motorbike':
        d = b % 10
        b //= 10
        n = b % 1000
        b //= 1000
        p = b % 100
        plate = '59%s%d %03d.%02d' % (letter, d, n, p)
    else:
        with_digit = b % 2
        b //= 2
        mid = ''
        if with_digit:
            mid = str(b % 10)
            b //= 10
        n = b % 1000
        b //= 1000
        p = b % 100
        plate = '51%s%s-%03d.%02d' % (letter, mid, n, p)
    k2 = rnd()
    color = COLORS[(k2 * 12) >> 32]
    k3 = rnd()
    owner_name = owners[(k3 * len(owners)) >> 32] if owners else 'Chủ xe %d' % slot
    k4 = rnd()
    owner_phone = '09%08d' % (k4 % 100000000)
    sc = next((s for s in SCENARIO_PLATES
               if s['slot'] == slot and s['gen'] == gen
               and (('motorbike' if s['vehicle_class'] == 'motorbike' else 'car') == fmt)), None)
    row = {'slot': slot, 'gen': gen, 'plate': sc['plate'] if sc else plate,
           'plate_format': fmt, 'vehicle_class': vehicle_class, 'color': color,
           'owner_name': owner_name, 'owner_phone': owner_phone,
           'is_scenario': bool(sc)}
    return row


def generate_all(slot_count=3600, gens=(0, 1, 2), owners=()):
    # Mỗi (slot, gen) có 2 bản ghi theo ĐỊNH DẠNG biển: 'motorbike' và 'car'.
    # bus/truck dùng định dạng 'car' (plateFor32 bản JS cũng vậy) nên lookup theo plate luôn thấy.
    rows = []
    for slot in range(slot_count):
        for gen in gens:
            for cls in ('motorbike', 'car'):
                rows.append(plate_for(slot, gen, cls, owners))
    return rows


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument('--out', default=os.path.join(os.path.dirname(__file__), 'plates.json'))
    ap.add_argument('--slots', type=int, default=3600)
    args = ap.parse_args()
    owners_path = os.path.join(os.path.dirname(__file__), 'owners.json')
    with open(owners_path, encoding='utf-8') as f:
        owners = json.load(f)
    rows = generate_all(args.slots, (0, 1, 2), owners)
    with open(args.out, 'w', encoding='utf-8') as f:
        json.dump(rows, f, ensure_ascii=False, separators=(',', ':'))
    print(json.dumps({'rows': len(rows), 'out': args.out}))


if __name__ == '__main__':
    main()
```

```python
import base64
import hashlib
import hmac
import json
import os
import time

import bcrypt
import jwt
from fastapi import HTTPException, Request

from app import config

LOGIN_HITS = {}      # ip -> [epoch, count]
LOGIN_LOCKS = {}     # ip -> locked_until_epoch


def hash_password(pw: str) -> str:
    return bcrypt.hashpw(pw.encode('utf-8'), bcrypt.gensalt(rounds=12)).decode('utf-8')


def verify_password(pw: str, hashed: str) -> bool:
    try:
        return bcrypt.checkpw(pw.encode('utf-8'), hashed.encode('utf-8'))
    except ValueError:
        return False


def load_or_create_secret() -> str:
    p = config.JWT_SECRET_PATH
    if p.exists():
        return p.read_text(encoding='utf-8').strip()
    secret = base64.b64encode(os.urandom(48)).decode('ascii')
    p.write_text(secret, encoding='utf-8')
    try:
        os.chmod(p, 0o600)
    except OSError:
        pass
    return secret


def make_token(user) -> str:
    now = int(time.time())
    payload = {'sub': user['username'], 'role': user['role'], 'iat': now, 'exp': now + config.JWT_TTL_S}
    return jwt.encode(payload, load_or_create_secret(), algorithm='HS256')


def decode_token(token: str) -> dict:
    try:
        return jwt.decode(token, load_or_create_secret(), algorithms=['HS256'])
    except jwt.PyJWTError:
        raise HTTPException(status_code=401, detail={'error': {'code': 'UNAUTHORIZED', 'message': 'Token không hợp lệ', 'detail': None}})


def bearer_of(request: Request) -> str:
    auth = request.headers.get('Authorization', '')
    if not auth.startswith('Bearer '):
        raise HTTPException(status_code=401, detail={'error': {'code': 'UNAUTHORIZED', 'message': 'Thiếu token', 'detail': None}})
    return auth[7:]


def require_role(*roles):
    def dep(request: Request):
        claims = decode_token(bearer_of(request))
        if claims.get('role') not in roles:
            raise HTTPException(status_code=403, detail={'error': {'code': 'FORBIDDEN', 'message': 'Không đủ quyền', 'detail': None}})
        request.state.claims = claims
        return claims
    return dep


class LoginLimiter:
    @staticmethod
    def check(ip: str):
        now = time.time()
        until = LOGIN_LOCKS.get(ip, 0)
        if now < until:
            raise HTTPException(status_code=429, detail={'error': {'code': 'RATE_LIMITED', 'message': 'Đăng nhập quá nhiều, thử lại sau 5 phút', 'detail': None}})
        window, count = LOGIN_HITS.get(ip, (now, 0))
        if now - window >= 60:
            window, count = now, 0
        count += 1
        LOGIN_HITS[ip] = (window, count)
        if count > config.LOGIN_RATE_LIMIT:
            LOGIN_LOCKS[ip] = now + config.LOGIN_LOCK_S
            raise HTTPException(status_code=429, detail={'error': {'code': 'RATE_LIMITED', 'message': 'Đăng nhập quá nhiều, thử lại sau 5 phút', 'detail': None}})
```

```python
# Hub WebSocket theo vai trò — auth bằng tin nhắn đầu {"type":"auth","token"} trong 5 s.
import asyncio
import json
import time

from app.auth import decode_token

SEQ = 0
CLIENTS = []   # {ws, role, username}


def _next_seq():
    global SEQ
    SEQ += 1
    return SEQ


async def ws_handler(websocket):
    role = None
    try:
        raw = await asyncio.wait_for(websocket.receive_text(), timeout=5)
        msg = json.loads(raw)
        if msg.get('type') != 'auth':
            await websocket.close(code=4401)
            return
        claims = decode_token(msg.get('token', ''))
        role = claims['role']
        CLIENTS.append({'ws': websocket, 'role': role, 'username': claims['sub']})
        while True:
            data = await websocket.receive_text()
            payload = json.loads(data)
            if 'event' not in payload:
                continue
            await broadcast(payload['event'], payload.get('data', {}), to_roles=None)
    except (asyncio.TimeoutError, Exception):
        pass
    finally:
        CLIENTS[:] = [c for c in CLIENTS if c['ws'] is not websocket]


async def broadcast(event: str, data: dict, to_roles=None):
    import datetime
    payload = json.dumps({'event': event, 't': datetime.datetime.utcnow().isoformat() + 'Z',
                          'seq': _next_seq(), 'data': data}, ensure_ascii=False)
    dead = []
    for c in CLIENTS:
        if to_roles is not None and c['role'] not in to_roles:
            continue
        try:
            await c['ws'].send_text(payload)
        except Exception:
            dead.append(c)
    for c in dead:
        CLIENTS[:] = [x for x in CLIENTS if x is not c]


def broadcast_sync(event: str, data: dict, to_roles=None):
    """Gọi từ endpoint đồng bộ — xếp lịch broadcast trên event loop đang chạy."""
    try:
        loop = asyncio.get_event_loop()
        if loop.is_running():
            loop.create_task(broadcast(event, data, to_roles))
    except RuntimeError:
        pass
```

```js
import { buildSignalPlan32 } from './signal-plan-32.js?v=32a';
import { createSignalController32 } from './traffic-controller-32.js?v=32a';

export const SMART_APP_32_VERSION = 'smart-app-32a';
export const SMART_BREAKER_S = Infinity;
export const BANNED_STRINGS = ['giả lập', 'mô phỏng', 'minh họa', 'illustrative', 'simulated', 'demo', 'fake'];
export const SMART_LABELS_32 = [
  { selector: '#regionTitle', from: 'Khu mô phỏng', to: 'Khu vực' },
  { selector: '#runFloodScenario span', from: 'Chạy mô phỏng 1D/2D', to: 'Chạy tính toán 1D/2D' },
  { selector: '#floodLockState', from: 'Không nạp được registry · mô phỏng đã khóa', to: 'Không nạp được registry · chế độ đã khóa' }
];

const escapeRe = s => s.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');

export function sanitizeStatusText32(text) {
  let out = String(text ?? '');
  for (const bad of BANNED_STRINGS) out = out.replace(new RegExp(escapeRe(bad), 'gi'), '');
  return out.replace(/\s{2,}/g, ' ').replace(/\s+([,.:;])/g, '$1').trim();
}

export function applySmartLabels32(root) {
  let n = 0;
  for (const item of SMART_LABELS_32) {
    const el = root.querySelector(item.selector);
    if (el && el.textContent === item.from) { el.textContent = item.to; n++; }
    else if (el && BANNED_STRINGS.some(b => el.textContent.toLowerCase().includes(b.toLowerCase()))) {
      el.textContent = sanitizeStatusText32(el.textContent); n++;
    }
  }
  return n;
}

export function lockRegions32(root) {
  for (const btn of root.querySelectorAll('[data-region]')) {
    const id = btn.dataset.region;
    if (id === 'bac') { btn.disabled = false; continue; }
    btn.disabled = true;
    btn.setAttribute('aria-disabled', 'true');
    btn.title = 'Khu vực này tạm khóa';
  }
  for (const panel of root.querySelectorAll('[data-classic-panel]')) panel.hidden = true;
}

export async function prepareSmart32({ graph }) {
  const res = await fetch('../smart-traffic/backend/app/seed/node_map.json');
  const nodeMap = await res.json();

}

  lockRegions32(root);
  const controller = createSignalController32({ net: trafficApi.net, sim: trafficApi.sim, mode });
  trafficApi.sim.attachController(controller);
  return { controller };
}
```

```js
// Đèn giao thông 3D + đếm ngược + cột camera cho cụm Hàng Xanh (M1).
import * as THREE_NS from 'three/webgpu';

export const SIGNALS_32_VERSION = 'signals-32a';
const PHASE_COLOR = { green: 0x2ecc71, yellow: 0xf1c40f, red: 0xe74c3c };
const IDLE_COLOR = 0x2a2f33;

export function lightColorOf(phase) {
  return phase === 'green' ? 'green' : phase === 'yellow' ? 'yellow' : 'red';
}

export function createTrafficSignals32({ scene, THREE = THREE_NS, controller, nodeMap, cameras }) {
  const group = new THREE.Group();
  group.name = 'traffic-signals-32';
  scene.add(group);
  const heads = [];   // {nodeIndex, groupIdx, lampMeshes:{green,yellow,red}, countdownMesh, texture, canvas}

  const makeLamp = (color, x, y) => {
    const mesh = new THREE.Mesh(new THREE.SphereGeometry(0.28, 12, 12),
      new THREE.MeshStandardMaterial({ color: IDLE_COLOR, emissive: IDLE_COLOR, emissiveIntensity: 0.2 }));
    mesh.position.set(x, y, 0);
    return mesh;
  };

  for (const node of nodeMap) {
    const pole = new THREE.Mesh(new THREE.CylinderGeometry(0.12, 0.16, 5.2, 10),
      new THREE.MeshStandardMaterial({ color: 0x555c60, roughness: 0.7 }));
    pole.position.set(node.x + 2.2, 2.6, node.z + 2.2);
    group.add(pole);
    for (const g of [0, 1]) {
      const box = new THREE.Mesh(new THREE.BoxGeometry(0.8, 2.1, 0.5),
        new THREE.MeshStandardMaterial({ color: 0x22262a, roughness: 0.8 }));
      box.position.set(node.x + 2.2 + g * 1.1, 4.6, node.z + 2.2);
      group.add(box);
      const lamps = {
        green: makeLamp(PHASE_COLOR.green, 0, 0.65),
        yellow: makeLamp(PHASE_COLOR.yellow, 0, 0),
        red: makeLamp(PHASE_COLOR.red, 0, -0.65)
      };
      for (const m of Object.values(lamps)) { box.add(m); }
      // Bảng đếm ngược: canvas 128×64 → texture.
      const canvas = document.createElement('canvas');
      canvas.width = 128; canvas.height = 64;
      const texture = new THREE.CanvasTexture(canvas);
      const countdown = new THREE.Mesh(new THREE.PlaneGeometry(1.2, 0.6),
        new THREE.MeshBasicMaterial({ map: texture, transparent: true }));
      countdown.position.set(node.x + 2.2 + g * 1.1, 3.2, node.z + 2.2);
      group.add(countdown);
      heads.push({ nodeIndex: node, groupIdx: g, lamps, countdown, canvas, texture });
    }
    // Cột camera + nón FOV (nếu node có camera).
    const cams = (cameras || []).filter(c => c.node_id === node.node_id);
    for (const cam of cams) {
      const column = new THREE.Mesh(new THREE.CylinderGeometry(0.1, 0.12, cam.height_m, 8),
        new THREE.MeshStandardMaterial({ color: 0x6b7276 }));
      column.position.set(cam.x, cam.height_m / 2, cam.z);
      group.add(column);
      const coneLen = cam.range_m;
      const cone = new THREE.Mesh(new THREE.ConeGeometry(Math.tan((cam.fov_deg * Math.PI / 180) / 2) * coneLen, coneLen, 16, 1, true),
        new THREE.MeshBasicMaterial({ color: 0x69a7ff, transparent: true, opacity: 0.12, side: THREE.DoubleSide, depthWrite: false }));
      const yaw = cam.yaw_deg * Math.PI / 180, pitch = cam.pitch_deg * Math.PI / 180;
      cone.position.set(cam.x + Math.cos(yaw) * coneLen / 2 * Math.cos(pitch), cam.height_m + Math.sin(pitch) * coneLen / 2, cam.z + Math.sin(yaw) * coneLen / 2 * Math.cos(pitch));
      cone.rotation.set(pitch - Math.PI / 2, 0, -yaw + Math.PI / 2);
      group.add(cone);
    }
  }

  const drawCountdown = (head, seconds) => {
    const ctx = head.canvas.getContext('2d');
    ctx.fillStyle = '#111417';
    ctx.fillRect(0, 0, 128, 64);
    ctx.fillStyle = '#f4f7f8';
    ctx.font = 'bold 44px Arial';
    ctx.textAlign = 'center';
    ctx.fillText(String(Math.max(0, Math.ceil(seconds))), 64, 48);
    head.texture.needsUpdate = true;
  };

  return {
    group,
    update(tSim) {
      for (const head of heads) {
        const nodeRow = head.nodeIndex;
        const state = controller.stateOf(node.node_id, tSim);   // stateOf nhận node_id (chuỗi) hoặc chỉ số nút
        const phaseInfo = state.phases.find(p => p.group === head.groupIdx) || state.phases[0];
        const lit = lightColorOf(phaseInfo.phase);
        for (const [name, mesh] of Object.entries(head.lamps)) {
          const on = name === lit;
          mesh.material.color.setHex(on ? PHASE_COLOR[name] : IDLE_COLOR);
          mesh.material.emissive.setHex(on ? PHASE_COLOR[name] : IDLE_COLOR);
          mesh.material.emissiveIntensity = on ? 1.2 : 0.15;
        }
        drawCountdown(head, phaseInfo.countdown_s);
      }
    },
    dispose() {
      scene.remove(group);
    }
  };
```

```js
// Render-to-texture camera theo registry cameras.json: khung stream + ảnh bằng chứng.
export const CAPTURE_32_VERSION = 'capture-32a';

export function projectToCamera32(x, z, cam) {
  const dx = x - cam.x, dz = z - cam.z;
  const distance = Math.hypot(dx, dz);
  const toPoint = Math.atan2(dz, dx);
  const yaw = cam.yaw_deg * Math.PI / 180;
  let d = toPoint - yaw;
  while (d > Math.PI) d -= 2 * Math.PI;
  while (d < -Math.PI) d += 2 * Math.PI;
  return { distance_m: distance, angle_rad: d };
}

export function createCameraCapture32({ renderer, scene, THREE, cameras }) {
  const rigs = new Map();   // camera_id → {camera, target, latest}
  for (const cam of cameras) {
    const camera = new THREE.PerspectiveCamera(cam.fov_deg, 16 / 9, 0.5, 3000);
    const yaw = cam.yaw_deg * Math.PI / 180, pitch = cam.pitch_deg * Math.PI / 180;
    camera.position.set(cam.x, cam.height_m, cam.z);
    camera.rotation.order = 'YXZ';
    camera.rotation.set(-pitch, -yaw + Math.PI / 2, 0);
    const target = new THREE.WebGLRenderTarget(1280, 720);
    rigs.set(cam.camera_id, { cam, camera, target, latest: null });
  }
  const renderTo = rig => {
    renderer.setRenderTarget(rig.target);
    renderer.render(scene, rig.camera);
    renderer.setRenderTarget(null);
    const w = 1280, h = 720;
    const pixels = new Uint8Array(w * h * 4);
    const read = renderer.readRenderTargetPixelsAsync
      ? renderer.readRenderTargetPixelsAsync(rig.target, 0, 0, w, h, pixels).then(() => pixels)
      : Promise.resolve().then(() => { renderer.readRenderTargetPixels(rig.target, 0, 0, w, h, pixels); return pixels; });
    return read.then(buf => {
      const canvas = document.createElement('canvas');
      canvas.width = w; canvas.height = h;
      const ctx = canvas.getContext('2d');
      const img = ctx.createImageData(w, h);
      // WebGL pixels đọc ngược chiều dọc → lật khi vẽ.
      for (let y = 0; y < h; y++) {
        const src = (h - 1 - y) * w * 4, dst = y * w * 4;
        img.data.set(buf.subarray(src, src + w * 4), dst);
      }
      ctx.putImageData(img, 0, 0);
      return { jpeg: canvas.toDataURL('image/jpeg', 0.8), w, h, canvas };
    });
  };
  return {
    captureFrame(cameraId) {
      const rig = rigs.get(cameraId);
      if (!rig) return null;
      return renderTo(rig).then(frame => {
        rig.latest = { jpeg: frame.jpeg, seq: (rig.latest?.seq ?? 0) + 1, t: Date.now() };
        return rig.latest;
      });
    },
    captureEvidence(cameraId, vehicleIndex, tSim) {
      const rig = rigs.get(cameraId);
      if (!rig) return { before: null, after: null, plate_crop: null };
      return renderTo(rig).then(before => renderTo(rig).then(after => {
        // Cắt ô biển số: hộp 320×120 ở 1/3 dưới khung after (vị trí xe = tâm ngang).
        const crop = document.createElement('canvas');
        crop.width = 320; crop.height = 120;
        const ctx = crop.getContext('2d');
        ctx.drawImage(after.canvas, after.w / 2 - 160, after.h * 0.62, 320, 120, 0, 0, 320, 120);
        return {
          before: { jpeg: before.jpeg, w: before.w, h: before.h, kind: 'before' },
          after: { jpeg: after.jpeg, w: after.w, h: after.h, kind: 'after' },
          plate_crop: { jpeg: crop.toDataURL('image/jpeg', 0.9), w: 320, h: 120, kind: 'plate_crop' }
        };
      }));
    },
    latestFrame: cameraId => rigs.get(cameraId)?.latest ?? null,
    setStream(cameraId, on, fps) {
      const rig = rigs.get(cameraId);
      if (!rig) return;
      if (on && !rig.timer) {
        rig.timer = setInterval(() => this.captureFrame(cameraId), 1000 / Math.max(1, fps || 2));
      } else if (!on && rig.timer) {
        clearInterval(rig.timer);
        rig.timer = null;
      }
    }
  };
}
```

```js
// Client trạm: REST/WS + hàng đợi IndexedDB, gửi lại khi có mạng (mục 8 đặc tả).
export const TRANSPORT_32_VERSION = 'transport-32a';
export const QUEUE_MAX = 2000;

export function createQueue32({ store }) {
  let items = store.get('queue') || [];
  const save = () => store.set('queue', items);
  return {
    push(record) {
      items.push(record);
      let dropped = 0;
      while (items.length > QUEUE_MAX) { items.shift(); dropped++; }
      save();
      return dropped;
    },
    shift() { const r = items.shift(); save(); return r; },
    size() { return items.length; },
    peekAll() { return items.slice(); }
  };
}

export function createTransport32({ baseUrl, wsUrl, token, stationId, queue, fetchImpl = fetch, wsImpl = null }) {
  let ws = null, backoffS = 1, timer = null, online = true;
  const listeners = [];
  const post = async (path, body) => {
    const res = await fetchImpl(baseUrl + path, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json', Authorization: `Bearer ${token}` },
      body: JSON.stringify(body)
    });
    return res.json ? res.json() : {};
  };
  const flush = async () => {
    while (queue.size()) {
      const rec = queue.peekAll()[0];
      try {
        await post('/api/v1/station/violations', rec);
        queue.shift();
        backoffS = 1;
      } catch (e) {
        online = false;
        backoffS = Math.min(30, backoffS * 2);
        return;
      }
    }
    online = true;
  };
  return {
    start() {
      if (wsImpl && wsUrl) {
        ws = new wsImpl(wsUrl);
        ws.onopen = () => ws.send(JSON.stringify({ type: 'auth', token }));
        ws.onmessage = ev => {
          try {
            const msg = JSON.parse(ev.data);
            for (const fn of listeners) fn(msg);
          } catch (e) { /* bỏ qua tin rác */ }
        };
      }
      timer = setInterval(() => {
        flush();
        post('/api/v1/station/heartbeat', {
          station_id: stationId, app_version: TRANSPORT_32_VERSION,
          t: new Date().toISOString(), uptime_s: Math.floor(process?.uptime?.() ?? 0), fps: 0, webgpu: true
        }).catch(() => {});
      }, backoffS * 1000);
    },
    stop() { if (timer) clearInterval(timer); if (ws) ws.close(); },
    sendViolation: async record => { queue.push(record); await flush(); },
    sendEvidence: async (violationId, kind, dataUrl) => post(`/api/v1/station/violations/${violationId}/evidence`, { kind, dataUrl }),
    sendMetrics: payload => post('/api/v1/station/metrics', payload).catch(() => {}),
    sendFrame: (cameraId, seq, jpeg_base64, w, h) => post('/api/v1/station/frames', { camera_id: cameraId, seq, t: new Date().toISOString(), jpeg_base64, w, h }).catch(() => {}),
    onCommand: fn => listeners.push(fn),
    connectionState: () => (online ? 'online' : 'offline'),
    flush
  };
}
```

```ts
const BASE = (import.meta as any).env?.VITE_API_BASE || 'http://127.0.0.1:8900/api/v1';

export function authHeaders(): Record<string, string> {
  const token = localStorage.getItem('smart_token');
  return token ? { Authorization: `Bearer ${token}` } : {};
}

export async function login(username: string, password: string) {
  const res = await fetch(`${BASE}/auth/login`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ username, password })
  });
  const body = await res.json();
  if (body.access_token) localStorage.setItem('smart_token', body.access_token);
  return body;
}

export async function listViolations(query: Record<string, string>) {
  const qs = new URLSearchParams(query).toString();
  const res = await fetch(`${BASE}/officer/violations?${qs}`, { headers: authHeaders() });
  return res.json();
}

export async function getViolation(id: string) {
  const res = await fetch(`${BASE}/officer/violations/${id}`, { headers: authHeaders() });
  return res.json();
}

export async function review(id: string, action: 'confirm' | 'dismiss' | 'needs-review', note: string) {
  const res = await fetch(`${BASE}/officer/violations/${id}/${action}`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json', ...authHeaders() },
    body: JSON.stringify({ note })
  });
  return res.json();
}

export async function getKpi() {
  const res = await fetch(`${BASE}/officer/kpi`, { headers: authHeaders() });
  return res.json();
}

export async function getStreams() {
  const res = await fetch(`${BASE}/officer/streams`, { headers: authHeaders() });
  return res.json();
}

export const streamUrl = (cameraId: string) => `${BASE}/media/stream/${cameraId}`;
```

```ts
export function connectSmartWs(token: string, onEvent: (msg: any) => void) {
  let ws: WebSocket | null = null;
  let backoff = 1000;
  const open = () => {
    ws = new WebSocket('ws://127.0.0.1:8900/ws/v1');
    ws.onopen = () => {
      backoff = 1000;
      ws?.send(JSON.stringify({ type: 'auth', token }));
    };
    ws.onmessage = ev => {
      try { onEvent(JSON.parse(ev.data)); } catch { /* bỏ qua tin rác */ }
    };
    ws.onclose = () => {
      setTimeout(open, backoff);
      backoff = Math.min(30000, backoff * 2);
    };
  };
  open();
  return () => ws?.close();
}
```
